model{
  ## Variance structure for spline coefficients
  for(i in 1:M_count){ # Method loop i
    mu_delta[i] <- 0
    sd_delta[i] ~ dt(0,1,1)T(0,) 
    sigma_delta[i,i] <- pow(sd_delta[i],2)
  }
    sigma_delta[1,2] <- rho[1]*sd_delta[1]*sd_delta[2]
    sigma_delta[1,3] <- rho[2]*sd_delta[1]*sd_delta[3]
    sigma_delta[1,4] <- rho[3]*sd_delta[1]*sd_delta[4]
    sigma_delta[1,5] <- rho[4]*sd_delta[1]*sd_delta[5]
    sigma_delta[2,1] <- rho[1]*sd_delta[1]*sd_delta[2]
    sigma_delta[2,3] <- rho[5]*sd_delta[2]*sd_delta[3]
    sigma_delta[2,4] <- rho[6]*sd_delta[2]*sd_delta[4]
    sigma_delta[2,5] <- rho[7]*sd_delta[2]*sd_delta[5]
    sigma_delta[3,1] <- rho[2]*sd_delta[1]*sd_delta[3]
    sigma_delta[3,2] <- rho[5]*sd_delta[2]*sd_delta[3]
    sigma_delta[3,4] <- rho[8]*sd_delta[3]*sd_delta[4]
    sigma_delta[3,5] <- rho[9]*sd_delta[3]*sd_delta[5]
    sigma_delta[4,1] <- rho[3]*sd_delta[1]*sd_delta[4]
    sigma_delta[4,2] <- rho[6]*sd_delta[2]*sd_delta[4]
    sigma_delta[4,3] <- rho[8]*sd_delta[3]*sd_delta[4]
    sigma_delta[4,5] <- rho[10]*sd_delta[4]*sd_delta[5]
    sigma_delta[5,1] <- rho[4]*sd_delta[1]*sd_delta[5]
    sigma_delta[5,2] <- rho[7]*sd_delta[2]*sd_delta[5]
    sigma_delta[5,3] <- rho[9]*sd_delta[3]*sd_delta[5]
    sigma_delta[5,4] <- rho[10]*sd_delta[4]*sd_delta[5]
    
    inv.sigma_delta[1:M_count,1:M_count] <- inverse(sigma_delta[,])

## Model Estimates
for(p in 1:P_count) { # province loop matched to C
  for(m in 1:M_count){ # method loop
    for (t in 1:n_years) { 
      z[m,p,t] <- alpha_pms[m,p] + inprod(B.ik[p,t,],beta.k[m,p,])
    } # end time loop
  } # end M loop 
} # end P loop


## Parameter Estimates 
  for(p in 1:P_count) { # province loop
    for(m in 1:M_count){ # method loop
      alpha_pms[m,p] ~ dnorm(alpha_cms[m,matchcountry[p]],tau_alpha_pms[m]) 
      beta.k[m,p,kstar[p]] <- 0
      for(j in 1:(kstar[p]-1)){ # before kstar[p] 
        beta.k[m,p,(kstar[p] - j)] <- beta.k[m,p,(kstar[p] - j)+1] - delta.k[m,p,(kstar[p] - j)]
      } # end K1 loop
      for(j in (kstar[p]+1):K){ # after kstar[p] 
        beta.k[m,p,j] <- beta.k[m,p,(j-1)] + delta.k[m,p,(j-1)]
      } # end K2 loop
    } # end m loop
    for(j in 1:H){
      delta.k[c(1:M_count),p,j] ~ dmnorm(mu_delta,inv.sigma_delta[,])
    } # end H loop
  } # end P loop

## Estimating all the Categories here (including total private)
  for(p in 1:P_count){ # province loop
    for(m in 1:M_count){ # method loop
      for (t in 1:n_years) { # years loop  
        P[1,m,p,t] <- 1/(1+exp(-(z[m,p,t]))) # modelling this as before assuming that z[m,p,t] is log(pi_public/pi_private)
        P[2,m,p,t] <- 1-P[1,m,p,t] # this then gives you the total private sector
        r[m,p,t] <- logit(P[2,m,p,t]) # for logit normal data model
      }
    }
  }

## Likelihood
  for (k in 1:n_obs) {
    inv.Sigma_y[1:2,1:2,k] <- inverse(Sigma_y[1:2,1:2,k])
    y[k, 1:2] ~ dmnorm(c(z[matchmethod[k],matchsubnat[k],matchyears[k]], r[matchmethod[k],matchsubnat[k],matchyears[k]]), inv.Sigma_y[1:2,1:2,k]) # Logit bivaraiate Normal data model
    }
    
## Priors
  for(m in 1:M_count){ # method loop
    for(c in 1:C_count) { # country loop
      alpha_cms[m,c] ~ dnorm(0,tau_alpha_cms[m])
    }
    tau_alpha_pms[m] <- sigma_alpha_pms[m]^-2
    sigma_alpha_pms[m] ~ dt(0,1,1)T(0,)
    tau_alpha_cms[m] <- sigma_alpha_cms[m]^-2
    sigma_alpha_cms[m] ~ dt(0,1,1)T(0,)
  }
}
