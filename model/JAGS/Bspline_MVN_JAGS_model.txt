model{
  for(p in 1:P_count){
    a0[1:M_count, p] = beta_c[1:M_count, matchcountry[p]] + err_a0[1:M_count, p]
    a[1:M_count, p, 1] = err_a[1:M_count, p, 1]
    for (i in 2:(num_knots-1)) {
      a[1:M_count, p, i] = a[1:M_count, p, i-1] + err_a[1:M_count, p, i] # penalising splines
    }
    for(m in 1:M_count){  
      a[m, p, num_knots] = -sum(a[m, p, 1:(num_knots-1)]) # hard sum to 0 constraint
      for(t in 1:num_years) {
        Y_hat[p, m, t] = a0[m, p] + inprod(B[p, t, 1:num_knots], a[m, p, 1:num_knots])
        P[p, m, 1, t] = exp(Y_hat[p, m, t])/(1+exp(Y_hat[p, m, t]))
        P[p, m, 2, t] = 1 - P[p, m, 1, t] 
      }
    }
  }

  # Priors
  
  for(m in 1:M_count){
    sigma_y[m] ~ dnorm(0, (1/2)^2)T(0,)
  }
  
  # Inverse-Wishart prior for spline params
  inv.Sigma.a[1:M_count,1:M_count] ~ dwish(Omega[1:M_count,1:M_count], M_count+1)

  # Inverse-Wishart prior for intercept province-level params
  inv.Sigma.a0[1:M_count,1:M_count] ~ dwish(Omega[1:M_count,1:M_count], M_count+1)
  
  # Inverse-Wishart prior for country-level params
  inv.Sigma.betac[1:M_count,1:M_count] ~ dwish(Omega[1:M_count,1:M_count], M_count+1)
  
  for(c in 1:C_count) { # country loop
    err_betac[1:M_count,c] ~ dmnorm(rep(0,M_count), inv.Sigma.betac[1:M_count,1:M_count])
    beta_c[1:M_count,c] <- rep(0,M_count) + err_betac[1:M_count,c]
  }
  
  for(p in 1:P_count) { # province loop
    err_a0[1:M_count,p] ~ dmnorm(rep(0,M_count), inv.Sigma.a0[1:M_count,1:M_count])
    for(k in 1:num_knots){
      err_a[1:M_count, p, k] ~ dmnorm(rep(0,M_count), inv.Sigma.a[1:M_count,1:M_count])
    }
  }

  # Likelihood
  for (k in 1:num_data) {
    Y[k] ~ dnorm(Y_hat[matchsubnat[k], matchmethod[k], matchyears[k]], sigma_y[matchmethod[k]])
    Y_sim[k] ~ dnorm(Y_hat[matchsubnat[k], matchmethod[k], matchyears[k]], sigma_y[matchmethod[k]])
  }
}
